const aggData = await User.aggregate([
      {
        $match: { _id: mongoose.Types.ObjectId(userID) },
      },
      {
        $unwind: "$notifications",
      },
      {
        $sort: { "notifications._id": -1 },
      },
      {
        $group: { _id: "$_id", notifications: { $push: "$notifications" } },
      },
      {
        $lookup: {
          from: "users",
          localField: "notifications.user",
          foreignField: "_id",
          as: "user",
        },
      },
      {
        $project: {
          "user.username": 1,
          "user.gender": 1,
          "user._id": 1,
          "user.avatar": 1,
          notifications: 1,
          notificationCount: {
            $size: {
              $filter: {
                input: "$notifications",
                as: "item",
                cond: { $eq: ["$$item.read", false] },
              },
            },
          },
        },
      },
    ]);
